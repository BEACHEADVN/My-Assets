name: Mod repo
on:
  repository_dispatch:
    types: [tracker_trigger]

permissions:
  contents: write

jobs:
  inject-steps:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout Repository
        uses: actions/checkout@main
        with:
          token: ${{ secrets.GH_TOKEN }}

      # Step xử lý kiểm tra và sao chép cấu trúc tài nguyên (Boot & Fonts chi tiết)
      - name: Sync Resources (Boot and Specific Fonts)
        if: ${{ github.event.client_payload.enable_sync_resources != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        shell: bash
        run: |
          BOOT_DIR="./bin/modfile/UpdateFile/Boot"
          FONT_DIR="./bin/modfile/UpdateFile/Fonts/HyperOS"
          
          NEED_BOOT=0
          NEED_FONT=0

          echo "🔍 Kiểm tra các file yêu cầu trong $BOOT_DIR..."
          if [ -f "$BOOT_DIR/bootanimation.zip" ] && [ -f "$BOOT_DIR/bootaudio.mp3" ] && [ -f "$BOOT_DIR/update.sh" ]; then
            echo "✅ Thư mục Boot đã có đầy đủ 3 file (bootanimation.zip, bootaudio.mp3, update.sh)."
          else
            echo "⚠️ Thiếu một hoặc nhiều file mục tiêu trong thư mục Boot. Đánh dấu cần tải bổ sung..."
            mkdir -p "$BOOT_DIR"
            NEED_BOOT=1
          fi

          echo "🔍 Kiểm tra các file font trong $FONT_DIR..."
          if [ -f "$FONT_DIR/Bauhaus.ttf" ] && [ -f "$FONT_DIR/product_fonts.list" ] && [ -f "$FONT_DIR/system_fonts.list" ]; then
            echo "✅ Đã tìm thấy đầy đủ 3 file font (Bauhaus.ttf, product_fonts.list, system_fonts.list)."
          else
            echo "⚠️ Thiếu một hoặc nhiều file font mục tiêu trong thư mục Fonts. Đánh dấu cần tải bổ sung..."
            mkdir -p "$FONT_DIR"
            NEED_FONT=1
          fi

          if [ "$NEED_BOOT" = "1" ] || [ "$NEED_FONT" = "1" ]; then
            echo "📥 Đang tải tạm thời repository My-Assets..."
            git clone --depth 1 https://x-access-token:${GH_TOKEN}@github.com/BEACHEADVN/My-Assets.git assets_temp
            
            if [ "$NEED_BOOT" = "1" ]; then
              echo "🚚 Sao chép file sang Boot..."
              cp -r assets_temp/ROM/boot/* "$BOOT_DIR/" 2>/dev/null || echo "Thư mục ROM/boot rỗng."
              
              if [ -f "assets_temp/ROM/repo/update-sh-boot.txt" ]; then
                echo "🔄 Đang sao chép và đổi tên update-sh-boot.txt thành update.sh..."
                cp "assets_temp/ROM/repo/update-sh-boot.txt" "$BOOT_DIR/update.sh"
                chmod +x "$BOOT_DIR/update.sh"
              else
                echo "⚠️ Không tìm thấy file update-sh-boot.txt trong assets_temp/ROM/repo/"
              fi
            fi

            if [ "$NEED_FONT" = "1" ]; then
              echo "🚚 Sao chép toàn bộ font từ ROM/font sang Fonts/HyperOS..."
              cp -r assets_temp/ROM/font/* "$FONT_DIR/" 2>/dev/null || echo "Thư mục ROM/font trống."
            fi

            rm -rf assets_temp
            echo "SHOULD_PUSH_RESOURCES=1" >> $GITHUB_ENV
          else
            echo "✅ Tất cả thư mục tài nguyên và file font đều đầy đủ. Bỏ qua tải."
            echo "SHOULD_PUSH_RESOURCES=0" >> $GITHUB_ENV
          fi

      - name: Inject Rclone Step
        if: ${{ github.event.client_payload.enable_rclone == true }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        run: |
          curl -s -H "Authorization: token $GH_TOKEN" -H "Accept: application/vnd.github.v3.raw" \
            -L "https://api.github.com/repos/BEACHEADVN/My-Assets/contents/ROM/repo/tai-rclone-tu-repo-private.txt?ref=main" \
            -o rclone.txt
          python3 - << 'EOF'
          import os
          build_path = ".github/workflows/build.yml"
          if os.path.exists(build_path):
              with open(build_path, "r", encoding="utf-8") as f:
                  build_lines = f.readlines()
              if any("Fetch Secret rclone.conf to Root Directory" in line for line in build_lines):
                  print("Step Rclone đã tồn tại, bỏ qua.")
                  exit(0)
              target_idx = -1
              for i, line in enumerate(build_lines):
                  if "- name: Set up Rclone" in line:
                      target_idx = i
                      break
              if target_idx != -1:
                  line_target = build_lines[target_idx]
                  indent_size = len(line_target) - len(line_target.lstrip())
                  base_indent = " " * indent_size
                  with open("rclone.txt", "r", encoding="utf-8") as f:
                      rclone_content = f.read()
                  formatted_lines = []
                  for sub_line in rclone_content.splitlines():
                      stripped = sub_line.strip()
                      if not stripped:
                          formatted_lines.append("")
                          continue
                      if stripped.startswith("- name:"):
                          formatted_lines.append(f"{base_indent}{stripped}")
                      elif stripped.startswith("env:") or stripped.startswith("run:") or stripped.startswith("id:") or stripped.startswith("shell:") or stripped.startswith("if:"):
                          formatted_lines.append(f"{base_indent}  {stripped}")
                      else:
                          formatted_lines.append(f"{base_indent}    {stripped}")
                  rclone_code = "\n".join(formatted_lines) + "\n\n"
                  build_lines.insert(target_idx, rclone_code)
                  with open(build_path, "w", encoding="utf-8") as f:
                      f.writelines(build_lines)
                  print("Đã chèn Rclone chuẩn cấu trúc thành công.")
          EOF

      - name: Inject Pixeldrain Step
        if: ${{ github.event.client_payload.enable_pixeldrain != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        run: |
          curl -s -H "Authorization: token $GH_TOKEN" -H "Accept: application/vnd.github.v3.raw" \
            -L "https://api.github.com/repos/BEACHEADVN/My-Assets/contents/ROM/repo/upload-pixeldrain.txt?ref=main" \
            -o pixeldrain.txt
          python3 - << 'EOF'
          import os
          build_path = ".github/workflows/build.yml"
          if os.path.exists(build_path):
              with open(build_path, "r", encoding="utf-8") as f:
                  build_lines = f.readlines()
              if any("Split Large File" in line for line in build_lines):
                  print("Step Pixeldrain đã tồn tại, bỏ qua.")
                  exit(0)
              target_idx = -1
              for i, line in enumerate(build_lines):
                  if "- name: Send success notification" in line:
                      target_idx = i
                      break
              if target_idx != -1:
                  line_target = build_lines[target_idx]
                  indent_size = len(line_target) - len(line_target.lstrip())
                  base_indent = " " * indent_size
                  with open("pixeldrain.txt", "r", encoding="utf-8") as f:
                      pixeldrain_content = f.read()
                  formatted_lines = []
                  for sub_line in pixeldrain_content.splitlines():
                      stripped = sub_line.strip()
                      if not stripped:
                          formatted_lines.append("")
                          continue
                      if stripped.startswith("- name:"):
                          formatted_lines.append(f"{base_indent}{stripped}")
                      elif stripped.startswith("env:") or stripped.startswith("run:") or stripped.startswith("id:") or stripped.startswith("shell:") or stripped.startswith("if:"):
                          formatted_lines.append(f"{base_indent}  {stripped}")
                      else:
                          formatted_lines.append(f"{base_indent}    {stripped}")
                  pixeldrain_code = "\n".join(formatted_lines) + "\n\n"
                  build_lines.insert(target_idx, pixeldrain_code)
                  with open(build_path, "w", encoding="utf-8") as f:
                      f.writelines(build_lines)
                  print("Đã chèn Pixeldrain chuẩn cấu trúc thành công.")
          EOF

      - name: Modify uploadROM.sh Script
        if: ${{ github.event.client_payload.enable_mod_uploadrom != false }}
        shell: bash
        run: |
          if [ -f ./uploadROM.sh ]; then
            echo "Modify uploadROM.sh script..."
            
            sed -i 's/ONEDRIVE_REMOTE=.*/ONEDRIVE_REMOTE="nothings-toolbuild"/g' ./uploadROM.sh
            
            if grep -q '--folder_id "1aJw0uWIMD7Yc1n4O1mjbI0YLHjRevfxl"' ./uploadROM.sh; then
              echo "Folder ID is already correct. Skipping update."
            else
              echo "Updating Folder ID..."
              sed -i 's/--folder_id "[^"]*"/--folder_id "1aJw0uWIMD7Yc1n4O1mjbI0YLHjRevfxl"/g' ./uploadROM.sh
            fi
            
            sed -i '/rm -rf \$work_dir\/out/d' ./uploadROM.sh
            sed -i '/rm -rf \$work_dir\/build/d' ./uploadROM.sh
            
            perl -0777 -pi -e 's/(if\s+\[\[\s+\$rom_os\s+==\s+"MIUI"\s+\]\];\s+then.*?upload\s+"Error uploading file to OneDrive: \$FILENAME"\s*\n\s*)exit 1(\s*\n\s*}\s*\n\s*else.*?upload\s+"Error uploading file to OneDrive: \$FILENAME"\s*\n\s*)exit 1/\1\2/sg' ./uploadROM.sh
            
            sed -i '/upload "Error uploading file to Google Drive"/{n; /exit 1/d}' ./uploadROM.sh
            
            echo "Script modification complete!"
          else
            echo "uploadROM.sh file not found!"
            exit 1
          fi

      - name: Fetch and Inject Font Logic from GitHub
        if: ${{ github.event.client_payload.enable_font_logic != false }}
        shell: bash
        run: |
          FONT_SCRIPT="./bin/modfile/UpdateFile/Fonts/update.sh"
          INJECT_URL="https://raw.githubusercontent.com/BEACHEADVN/My-Assets/main/ROM/repo/mod-font-update-sh.txt"
          
          if [ -f "$FONT_SCRIPT" ]; then
            if grep -q "copy_bauhaus_from_list" "$FONT_SCRIPT"; then
              echo "✅ Logic xử lý font list đã tồn tại trong update.sh, bỏ qua chèn để tránh trùng lặp."
            else
              echo "📥 Đang tải đoạn code chèn từ GitHub..."
              curl -sL "$INJECT_URL" > /tmp/font_inject.txt
              
              if [ -s /tmp/font_inject.txt ]; then
                echo "📝 Tiến hành chèn code vào cuối file update.sh..."
                sed -i -e '$a\' "$FONT_SCRIPT"
                cat /tmp/font_inject.txt >> "$FONT_SCRIPT"
                rm -f /tmp/font_inject.txt
                chmod +x "$FONT_SCRIPT"
                echo "✅ Đã chèn code vào cuối file update.sh (Fonts) thành công!"
              else
                echo "❌ Tải nội dung từ GitHub thất bại hoặc file trống!"
              fi
            fi
          else
            echo "⚠️ Không tìm thấy file gốc tại $FONT_SCRIPT để chỉnh sửa."
          fi

      - name: Inject Update Toolbuild Status Step
        if: ${{ github.event.client_payload.enable_update_status != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        run: |
          curl -s -H "Authorization: token $GH_TOKEN" -H "Accept: application/vnd.github.v3.raw" \
            -L "https://api.github.com/repos/BEACHEADVN/My-Assets/contents/ROM/repo/step-cap-nhat-trang-thai-toolbuild.txt?ref=main" \
            -o status_step.txt
          python3 - << 'EOF'
          import os
          build_path = ".github/workflows/build.yml"
          if os.path.exists(build_path):
              with open(build_path, "r", encoding="utf-8") as f:
                  build_lines = f.readlines()
              
              if any("Update Status to My-Assets" in line for line in build_lines):
                  print("Step Update Toolbuild Status đã tồn tại, bỏ qua.")
                  exit(0)
                  
              target_idx = -1
              for i, line in enumerate(build_lines):
                  if "- name: Upload to Pixeldrain" in line:
                      target_idx = i
                      break
                      
              if target_idx != -1:
                  line_target = build_lines[target_idx]
                  indent_size = len(line_target) - len(line_target.lstrip())
                  base_indent = " " * indent_size
                  
                  insert_idx = target_idx + 1
                  while insert_idx < len(build_lines):
                      next_line = build_lines[insert_idx].strip()
                      if build_lines[insert_idx].startswith(base_indent + "-"):
                          break
                      if next_line and not build_lines[insert_idx].startswith(base_indent):
                          break
                      insert_idx += 1
                  with open("status_step.txt", "r", encoding="utf-8") as f:
                      status_content = f.read()
                      
                  formatted_lines = []
                  for sub_line in status_content.splitlines():
                      stripped = sub_line.strip()
                      if not stripped:
                          formatted_lines.append("")
                          continue
                      if stripped.startswith("- name:"):
                          formatted_lines.append(f"{base_indent}{stripped}")
                      elif stripped.startswith("env:") or stripped.startswith("run:") or stripped.startswith("id:") or stripped.startswith("shell:") or stripped.startswith("if:"):
                          formatted_lines.append(f"{base_indent}  {stripped}")
                      else:
                          formatted_lines.append(f"{base_indent}    {stripped}")
                          
                  status_code = "\n".join(formatted_lines) + "\n\n"
                  build_lines.insert(insert_idx, status_code)
                  
                  with open(build_path, "w", encoding="utf-8") as f:
                      f.writelines(build_lines)
                  print("Đã chèn step Cập nhật trạng thái Toolbuild chuẩn cấu trúc thành công.")
              else:
                  print("⚠️ Không tìm thấy step '- name: Upload to Pixeldrain' trong file build.yml.")
          EOF

      - name: Inject Packages into build.yml
        if: ${{ github.event.client_payload.enable_inject_packages != false }}
        run: |
          python3 - << 'EOF'
          import os
          build_path = ".github/workflows/build.yml"
          if os.path.exists(build_path):
              with open(build_path, "r", encoding="utf-8") as f:
                  lines = f.readlines()
              
              if any("pip3 install lz4" in line for line in lines):
                  print("✅ Dòng 'sudo pip3 install lz4' đã tồn tại trong build.yml, bỏ qua.")
                  exit(0)
                  
              start_idx = -1
              for i, line in enumerate(lines):
                  if "Install tools and send start notification" in line:
                      start_idx = i
                      break
              
              if start_idx != -1:
                  print("📝 Tìm thấy step 'Install tools and send start notification', đang kiểm tra và chèn...")
                  
                  target_insert_idx = -1
                  for i in range(start_idx, len(lines)):
                      if i > start_idx and lines[i].strip().startswith("- name:"):
                          break
                      if "sudo pip3 install setuptools" in lines[i] or "sudo pip3 install ConfigObj" in lines[i]:
                          target_insert_idx = i
                  
                  if target_insert_idx == -1:
                      for i in range(start_idx, len(lines)):
                          if "python notify.py start" in lines[i]:
                              target_insert_idx = i - 1
                              break

                  if target_insert_idx != -1:
                      line_target = lines[target_insert_idx]
                      indent_size = len(line_target) - len(line_target.lstrip())
                      step_indent = " " * indent_size
                      
                      lines.insert(target_insert_idx + 1, f"{step_indent}sudo pip3 install lz4 2>&1 | tee -a build_action.log\n")
                      
                      with open(build_path, "w", encoding="utf-8") as f:
                          f.writelines(lines)
                      print("✅ Đã chèn chính xác 'sudo pip3 install lz4' vào block Install tools!")
                  else:
                      print("⚠️ Không tìm thấy vị trí thích hợp trong block Install tools.")
              else:
                  print("⚠️ Không tìm thấy step 'Install tools and send start notification' trong build.yml.")
          EOF

      - name: Modify packROM.sh Script
        if: ${{ github.event.client_payload.enable_mod_packrom != false }}
        shell: bash
        run: |
          if [ -f ./packROM.sh ]; then
            echo "📝 Đang tiến hành xóa flag --sparse trong packROM.sh..."
            sed -i 's/mkfs.erofs \(.*\)--sparse \(.*\)/mkfs.erofs \1\2/g' ./packROM.sh
            echo "✅ Đã loại bỏ flag hoàn tất!"
          else
            echo "⚠️ Không tìm thấy file packROM.sh để chỉnh sửa."
          fi

      - name: Check and Sync vbpatcher.py
        if: ${{ github.event.client_payload.enable_sync_vbpatcher != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        shell: bash
        run: |
          VBPATCHER_PATH="./bin/vbpatcher.py"
          RAW_URL="https://raw.githubusercontent.com/BEACHEADVN/My-Assets/main/ROM/repo/vbpatcher.py"

          mkdir -p "./bin"

          if [ -f "$VBPATCHER_PATH" ] && grep -q "^#chinh-sua-vbpatcher" "$VBPATCHER_PATH"; then
            echo "✅ File bin/vbpatcher.py đã chứa '#chinh-sua-vbpatcher' ở đầu, bỏ qua."
          else
            echo "📥 Đang tải file vbpatcher.py mới từ My-Assets..."
            curl -s -H "Authorization: token $GH_TOKEN" -L "$RAW_URL" -o "$VBPATCHER_PATH"
            echo "✅ Đã cập nhật bin/vbpatcher.py thành công!"
          fi

      - name: Check and Sync DISABLEavb.sh
        if: ${{ github.event.client_payload.enable_sync_disableavb != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        shell: bash
        run: |
          SCRIPT_PATH="./bin/package/DISABLE_AVB/DISABLEavb.sh"
          RAW_URL="https://raw.githubusercontent.com/BEACHEADVN/My-Assets/main/ROM/repo/DISABLEavb.sh"

          mkdir -p "./bin/package/DISABLE_AVB"

          if [ -f "$SCRIPT_PATH" ] && grep -q "^#chinh-sua-DISABLEavb" "$SCRIPT_PATH"; then

          else
            echo "📥 Đang tải file DISABLEavb.sh mới từ My-Assets..."

      - name: Check and Sync fetchINFO.sh
        if: ${{ github.event.client_payload.enable_sync_fetchinfo != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        shell: bash
        run: |
          SCRIPT_PATH="./bin/ddevice/fetchINFO.sh"
          RAW_URL="https://raw.githubusercontent.com/BEACHEADVN/My-Assets/main/ROM/repo/fetchINFO.sh"

          mkdir -p "./bin/ddevice"

          if [ -f "$SCRIPT_PATH" ] && grep -q "^#xu-ly-device_code" "$SCRIPT_PATH"; then

          else
            echo "📥 Đang tải file fetchINFO.sh mới từ My-Assets..."

            chmod +x "$SCRIPT_PATH"
            echo "✅ Đã cập nhật fetchINFO.sh thành công!"
          fi

      - name: Sync unique lines from debloat-list.txt to APPLIST.txt
        if: ${{ github.event.client_payload.enable_sync_debloat != false }}
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        shell: bash
        run: |
          TARGET_FILE="./bin/ddevice/DEBLOAT/APPLIST.txt"
          TEMP_REMOTE="/tmp/debloat-list.txt"
          RAW_URL="https://raw.githubusercontent.com/BEACHEADVN/My-Assets/main/ROM/repo/debloat-list.txt"

          mkdir -p "./bin/ddevice/DEBLOAT"

          [ ! -f "$TARGET_FILE" ] && touch "$TARGET_FILE"

          curl -s -H "Authorization: token $GH_TOKEN" -L "$RAW_URL" -o "$TEMP_REMOTE"

          ADDED_COUNT=0

          while IFS= read -r line || [ -n "$line" ]; do
            line_clean=$(echo "$line" | tr -d '\r\n')
            [ -z "$line_clean" ] && continue

            if ! grep -qxF "$line_clean" "$TARGET_FILE"; then
              echo "$line_clean" >> "$TARGET_FILE"
              echo "➕ Thêm ứng dụng mới: $line_clean"
              ADDED_COUNT=$((ADDED_COUNT + 1))
            fi
          done < "$TEMP_REMOTE"

          rm -f "$TEMP_REMOTE"

          if [ $ADDED_COUNT -eq 0 ]; then
            echo "✅ Tất cả các dòng trong debloat-list.txt đều đã có sẵn trong APPLIST.txt."
          else
            echo "🎉 Đã thêm thành công $ADDED_COUNT dòng mới vào APPLIST.txt!"
          fi

      - name: Push changes to Repository
        env:
          GH_TOKEN: ${{ secrets.GH_TOKEN }}
        run: |
          git config --local user.email "github-actions[bot]@users.noreply.github.com"
          git config --local user.name "github-actions[bot]"
          
          git add .github/workflows/build.yml uploadROM.sh ./bin/modfile/UpdateFile/Fonts/update.sh packROM.sh ./bin/vbpatcher.py ./bin/package/DISABLE_AVB/DISABLEavb.sh ./bin/ddevice/fetchINFO.sh ./bin/ddevice/DEBLOAT/APPLIST.txt
          
          if [ "${{ env.SHOULD_PUSH_RESOURCES }}" = "1" ]; then
            git add ./bin/modfile/UpdateFile/Boot/ ./bin/modfile/UpdateFile/Fonts/HyperOS/
          fi
          
          git commit -m "Automated: Sync steps, fetchINFO.sh and update APPLIST.txt" || echo "Không có thay đổi để commit"
          
          git remote set-url origin "https://x-access-token:${GH_TOKEN}@github.com/${{ github.repository }}.git"
          git push origin main
